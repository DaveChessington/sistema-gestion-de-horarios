// Actualizar calendario de forma asíncrona sin recargar la página
async function fetchScheduleAsync(filters = {}) {
    const params = new URLSearchParams(filters);
    const response = await fetch(`/api/v1/horarios?${params.toString()}`);
    const json = await response.json();
    renderCalendarDOM(json.data); // Actualiza solo el contenedor del calendario
}
